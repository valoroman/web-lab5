curl -sS -D - -o cache/curl-418.jpg http://127.0.0.1:3000/418

# PUT /200 and verify the saved JPEG
curl -i -X PUT -H 'Content-Type: image/jpeg' --data-binary '@cache/curl-418.jpg' http://127.0.0.1:3000/200
curl -sS -D - -o cache/curl-200-after-put.jpg http://127.0.0.1:3000/200
cmp cache/curl-418.jpg cache/curl-200-after-put.jpg && echo "JPEG files match"
# DELETE /200: remove the image and repeat
curl -i -X DELETE http://127.0.0.1:3000/200
curl -i -X DELETE http://127.0.0.1:3000/200
